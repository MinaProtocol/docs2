  hash: TransactionHash!

  isDelegation: Boolean! @deprecated
  kind: UserCommandKind!
  memo: String!
  nonce: Int!
  receiver: Account!
  source: Account!
  to: PublicKey! @deprecated
  toAccount: Account! @deprecated

}
